这是我自己第一开始想到的方法，但是这个单项映射是有问题的

```cpp
class Solution {
public:
    bool isIsomorphic(string s, string t) {
        if (s.length() != t.length()) return false;

        unordered_map<char, char> s_to_t;

        for (int i = 0; i < s.length(); i++) {
            char ch_s = s[i];
            char ch_t = t[i];

            if (s_to_t.find(ch_s) != s_to_t.end()) {
                if (s_to_t[ch_s] != ch_t) {
                    return false;
                }
            } else {
                s_to_t[ch_s] = ch_t;
            }
        }
        return true;
    }
};
```

比如说没有通过测试案例：

```text
s="babc"
t="dada"
```

实际上必须要考虑双向映射，然后数组访问速度更快：

```cpp
class Solution {
public:
    bool isIsomorphic(string s, string t) {
        if (s.length() != t.length()) return false;

        // 使用数组代替哈希表，提高效率
        char s_to_t[256] = {0};  // ASCII字符映射
        char t_to_s[256] = {0};

        for (int i = 0; i < s.length(); i++) {
            char ch_s = s[i], ch_t = t[i];

            // 检查s到t的映射
            if (s_to_t[ch_s] != 0 && s_to_t[ch_s] != ch_t) {
                return false;
            }

            // 检查t到s的映射
            if (t_to_s[ch_t] != 0 && t_to_s[ch_t] != ch_s) {
                return false;
            }

            // 建立映射
            s_to_t[ch_s] = ch_t;
            t_to_s[ch_t] = ch_s;
        }

        return true;
    }
};
```
