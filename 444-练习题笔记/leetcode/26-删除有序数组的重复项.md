```cpp
class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int head = 1, curr = 1;
        int head_val = nums[0];
        while (curr < nums.size()) {
            if (head_val != nums[curr]) {
                nums[head] = nums[curr];
                head_val = nums[head];
                head++;
            }
            curr++;
        }
        return head;
    }
};
```
