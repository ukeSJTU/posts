class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int head = 0, tail = nums.size() - 1;
        while (head <= tail) {
            if (nums[head] == val) {
                nums[head] = nums[tail];
                tail--;
            } else {
                head++;
            }
        }
        return head;
    }
};
