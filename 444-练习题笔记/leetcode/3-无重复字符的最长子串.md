```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_set<char> occ;
        int res = 0;
        for (int l = 0, r = 0; r < s.size(); r++) {
            char ch = s[r];
            while(occ.count(ch)) {
                occ.erase(s[l]);
                l++;
            }
            occ.insert(ch);
            res = max(res, r - l + 1);
        }
        return res;
    }
};
```
