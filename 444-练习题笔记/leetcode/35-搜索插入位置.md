```cpp
class Solution {
public:
    int searchHelper(vector<int>& nums, int target, int left, int right) {
        // 递归终止条件：找到插入位置
        if (left > right) {
            return left;
        }

        int mid = left + (right - left) / 2;  // 防止溢出

        if (nums[mid] == target) {
            return mid;
        } else if (nums[mid] > target) {
            return searchHelper(nums, target, left, mid - 1);
        } else {
            return searchHelper(nums, target, mid + 1, right);
        }
    }

    int searchInsert(vector<int>& nums, int target) {
        return searchHelper(nums, target, 0, nums.size() - 1);
    }
};
```

更简洁的写法

```cpp
class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int left = 0;
        int right = nums.size();  // 注意这里是 size()，不是 size()-1

        while (left < right) {
            int mid = left + (right - left) / 2;

            if (nums[mid] < target) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }

        return left;
    }
};
```
